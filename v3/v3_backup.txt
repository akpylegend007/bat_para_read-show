static uint16_t be16(const uint8_t *p) {
  return (static_cast<uint16_t>(p[0]) << 8) | p[1];
}
static int16_t signedBe16(const uint8_t *p) { return static_cast<int16_t>(be16(p)); }
static uint32_t be32(const uint8_t *p) {
  return (static_cast<uint32_t>(p[0]) << 24) | (static_cast<uint32_t>(p[1]) << 16) |
         (static_cast<uint32_t>(p[2]) << 8) | p[3];
}

static bool telemetryChanged(const Telemetry &a, const Telemetry &b) {
  return a.valid != b.valid || a.voltage != b.voltage || a.current != b.current ||
         a.remainingAh != b.remainingAh || a.designAh != b.designAh || a.cycles != b.cycles ||
         a.temperatureC != b.temperatureC || a.soc != b.soc || a.protection != b.protection ||
         a.balanceMaskLow != b.balanceMaskLow || a.balanceMaskHigh != b.balanceMaskHigh ||
         a.chargeFet != b.chargeFet || a.dischargeFet != b.dischargeFet || a.rssi != b.rssi;
}

static Telemetry snapshotTelemetry() {
  Telemetry copy;
  xSemaphoreTake(telemetryMutex, portMAX_DELAY);
  copy = telemetry;
  xSemaphoreGive(telemetryMutex);
  return copy;
}

static const char *modeText(const Telemetry &d) {
  if (d.current > 0.15f) return "CHARGING";
  if (d.current < -0.15f) return "DISCHARGING";
  return "IDLE";
}
static String alarmText(const Telemetry &d) {
  if (d.protection == 0) return "CLEAR";
  String out;
  if (d.protection & 0x0001) out += "CELL OV ";
  if (d.protection & 0x0002) out += "CELL UV ";
  if (d.protection & 0x0004) out += "PACK OV ";
  if (d.protection & 0x0008) out += "PACK UV ";
  if (d.protection & 0x0010) out += "CHG OC ";
  if (d.protection & 0x0020) out += "DSG OC ";
  if (d.protection & 0x0040) out += "SHORT ";
  if (d.protection & 0x0F00) out += "TEMP ";
  return out.length() ? out : "BMS FAULT";
}
static uint8_t balanceCount(const Telemetry &d) {
  return __builtin_popcount(d.balanceMaskLow) + __builtin_popcount(d.balanceMaskHigh);
}
static String balanceCellsText(const Telemetry &d) {
  String out;
  for (uint8_t cell = 0; cell < 32; ++cell) if (d.balanceMaskLow & (1UL << cell)) out += (out.length() ? " " : "") + String("C") + String(cell + 1);
  for (uint8_t cell = 0; cell < 32; ++cell) if (d.balanceMaskHigh & (1UL << cell)) out += (out.length() ? " " : "") + String("C") + String(cell + 33);
  return out.length() ? out : "NONE";
}

static bool parseJbdFrame(const uint8_t *frame, size_t length) {
  if (length < 34 || frame[0] != 0xDD || frame[1] != 0x03 || frame[length - 1] != 0x77) return false;
  const uint8_t payloadLength = frame[3];
  if (payloadLength < 27 || length != static_cast<size_t>(payloadLength) + 7) return false;
  const uint8_t *p = &frame[4];
  Telemetry next = snapshotTelemetry();
  next.voltage = be16(p) / 100.0f; next.current = signedBe16(p + 2) / 100.0f;
  next.remainingAh = be16(p + 4) / 100.0f; next.designAh = be16(p + 6) / 100.0f; next.cycles = be16(p + 8);
  // JBD basic-information (0x03) payload layout: balance mask 12..15,
  // protection 16..17, firmware 18, RSOC 19, FET state 20, NTC count 21,
  // then 0.1 K temperature values beginning at 22. Do not treat the first
  // temperature byte as SOC: that produces a plausible-looking but wrong %.
  next.balanceMaskLow = be32(p + 12);
  next.balanceMaskHigh = 0;  // Basic-info 0x03 carries one 32-cell balance mask.
  next.protection = be16(p + 16);
  next.soc = p[19];
  next.chargeFet = (p[20] & 0x01) != 0;
  next.dischargeFet = (p[20] & 0x02) != 0;
  if (p[21] > 0 && payloadLength >= 24) next.temperatureC = static_cast<int16_t>(be16(p + 22) / 10) - 273;
  next.valid = true;
  xSemaphoreTake(telemetryMutex, portMAX_DELAY);
  bool changed = telemetryChanged(telemetry, next); telemetry = next;
  xSemaphoreGive(telemetryMutex);
  if (changed) dataDirty = true;
  Serial.printf("JBD data: SOC=%u V=%.2f I=%.2f\n", next.soc, next.voltage, next.current);
  return true;
}
static void notificationCallback(NimBLERemoteCharacteristic *, uint8_t *payload, size_t length, bool) {
  for (size_t i = 0; i < length; ++i) {
    if (rxLength == 0 && payload[i] != 0xDD) continue;
    if (rxLength >= sizeof(rxFrame)) { rxLength = 0; continue; }
    rxFrame[rxLength++] = payload[i];
    if (rxLength >= 4) {
      size_t expected = static_cast<size_t>(rxFrame[3]) + 7;
      if (expected > sizeof(rxFrame)) rxLength = 0;
      else if (rxLength == expected) { parseJbdFrame(rxFrame, rxLength); rxLength = 0; }
    }
  }
}
class BatteryAdvertisedCallbacks : public NimBLEScanCallbacks {
  void onResult(const NimBLEAdvertisedDevice *device) override {
    if (!device->haveName()) return;
    std::string found = device->getName();
    if (found.find(BATTERY_NAME_MATCH) == std::string::npos) return;
    batteryAddress = device->getAddress(); haveBatteryAddress = true; scanRunning = false;
    Serial.printf("battery found: %s (%s)\n", found.c_str(), batteryAddress.toString().c_str());
    NimBLEDevice::getScan()->stop();
  }
};
class BatteryClientCallbacks : public NimBLEClientCallbacks {
  void onConnect(NimBLEClient *) override { batteryConnected = true; frameDirty = true; Serial.println("BLE connected"); }
  void onDisconnect(NimBLEClient *, int reason) override {
    batteryConnected = false; commandChar = nullptr; haveBatteryAddress = false; frameDirty = true;
    Serial.printf("BLE disconnected, reason=%d\n", reason);
  }
};
static bool findAndConnectBattery() {
  if (!haveBatteryAddress) {
    if (!scanRunning) {
      NimBLEScan *scan = NimBLEDevice::getScan(); scan->setScanCallbacks(new BatteryAdvertisedCallbacks(), true);
      scan->setInterval(80); scan->setWindow(60); scan->setActiveScan(true); scanRunning = true;
      Serial.printf("BLE scan: continuously looking for name containing '%s'\n", BATTERY_NAME_MATCH);
      // A finite scan used to end after five seconds while scanRunning remained true,
      // preventing every later connection attempt. Keep scanning until the matching BMS is found.
      scan->start(0, false);
    }
    return false;
  }
  if (client) { NimBLEDevice::deleteClient(client); client = nullptr; }
  Serial.println("connecting to matched battery"); client = NimBLEDevice::createClient(); client->setClientCallbacks(new BatteryClientCallbacks(), true);
  if (!client->connect(batteryAddress)) { Serial.println("BLE connection failed"); NimBLEDevice::deleteClient(client); client = nullptr; haveBatteryAddress = false; return false; }
  NimBLERemoteService *service = client->getService("0000ff00-0000-1000-8000-00805f9b34fb");
  if (!service) { Serial.println("FF00 service missing"); client->disconnect(); return false; }
  NimBLERemoteCharacteristic *notifyChar = service->getCharacteristic("0000ff01-0000-1000-8000-00805f9b34fb");
  commandChar = service->getCharacteristic("0000ff02-0000-1000-8000-00805f9b34fb");
  if (!notifyChar || !commandChar) { Serial.println("FF01 or FF02 characteristic missing"); client->disconnect(); commandChar = nullptr; return false; }
  if (!notifyChar->subscribe(true, notificationCallback)) { Serial.println("FF01 notification subscription failed"); client->disconnect(); commandChar = nullptr; return false; }
  Serial.println("BLE link subscribed; read-only JBD polling begins"); return true;
}
static void pollBattery() {
  static const uint8_t request[] = {0xDD, 0xA5, 0x03, 0x00, 0xFF, 0xFD, 0x77};
  if (!client || !client->isConnected() || !commandChar) return;
  Telemetry now = snapshotTelemetry(); now.rssi = client->getRssi();
  xSemaphoreTake(telemetryMutex, portMAX_DELAY); bool changed = telemetry.rssi != now.rssi; telemetry.rssi = now.rssi; xSemaphoreGive(telemetryMutex); if (changed) dataDirty = true;
  if (!commandChar->writeValue(request, sizeof(request), false)) Serial.println("JBD read request write failed");
}

static const char OTA_FORM[] PROGMEM = R"HTML(<!doctype html><html><body style='background:#000;color:#ffe600;font-family:monospace'><h2>PAC-BATTERY UPDATE</h2><p>ESP32 firmware only. The BMS is never written.</p><form method='POST' action='/update' enctype='multipart/form-data'><input type='file' name='firmware' accept='.bin' required><button type='submit'>Install update</button></form></body></html>)HTML";
static void startOtaPortal() {
  WiFi.mode(WIFI_AP); WiFi.softAP(OTA_AP_NAME, OTA_AP_PASSWORD);
  otaServer.on("/", HTTP_GET, []() { otaServer.send_P(200, "text/html", OTA_FORM); });
  otaServer.on("/update", HTTP_POST, []() {
    otaServer.sendHeader("Connection", "close"); otaServer.send(200, "text/plain", Update.hasError() ? "Update failed" : "Update complete. Restarting.");
    delay(500); ESP.restart();
  }, []() {
    HTTPUpload &upload = otaServer.upload();
    if (upload.status == UPLOAD_FILE_START) Update.begin(UPDATE_SIZE_UNKNOWN);
    else if (upload.status == UPLOAD_FILE_WRITE) Update.write(upload.buf, upload.currentSize);
    else if (upload.status == UPLOAD_FILE_END) Update.end(true);
  });
  otaServer.begin(); otaActive = true; frameDirty = true; Serial.println("OTA AP active at 192.168.4.1");
}
static void stopOtaPortal() { otaServer.stop(); WiFi.softAPdisconnect(true); WiFi.mode(WIFI_OFF); otaActive = false; frameDirty = true; Serial.println("OTA AP stopped"); }

static void serviceTouch() {
  static bool tracking = false;
  static uint16_t startX = 0, latestX = 0;
  static uint32_t startedMs = 0, lastDownMs = 0, lastGestureMs = 0;
  
  if (Touch_interrupts) {
    Touch_interrupts = false;
    Touch_Read_Data();
  }
  
  uint16_t x[CST328_LCD_TOUCH_MAX_POINTS];
  uint16_t y[CST328_LCD_TOUCH_MAX_POINTS];
  uint16_t strength[CST328_LCD_TOUCH_MAX_POINTS];
  uint8_t point_num = 0;
  
  bool down = (Touch_Get_XY(x, y, strength, &point_num, CST328_LCD_TOUCH_MAX_POINTS) && point_num > 0);
  uint32_t now = millis();
  
  if (down) {
    if (!tracking) { 
      // Start of a new touch
      tracking = true; 
      startX = latestX = x[0]; 
      startedMs = now; 
    } else {
      // Continuing touch
      latestX = x[0];
    }
    lastDownMs = now; // update last seen touch time
  } else if (tracking) {
    // We are tracking, but no touch right now. Debounce the release!
    if (now - lastDownMs > 50) {
      // 50ms without touch -> confirmed release
      tracking = false; 
      
      if (now - lastGestureMs < 250) return; // Cooldown between actions
      
      int16_t dx = (int16_t)latestX - (int16_t)startX;
      uint32_t duration = lastDownMs - startedMs;
      
      if (duration >= 1200) {
        // Long press -> OTA toggle
        otaToggleRequested = true; 
        lastGestureMs = now;
      } else if (abs(dx) >= 35) { 
        // Horizontal swipe -> change page
        page = dx < 0 ? (page + 1) % 4 : (page + 3) % 4; 
        frameDirty = true; 
        lastGestureMs = now;
        Serial.printf("touch swipe: dx=%d, page=%u\n", dx, page + 1); 
      } else if (duration < 500 && abs(dx) < 20) {
        // Short tap without much movement -> change page based on screen half
        if (startX < (SCREEN_W / 2)) {
          // Tap left half -> previous page
          page = (page + 3) % 4;
        } else {
          // Tap right half -> next page
          page = (page + 1) % 4;
        }
        frameDirty = true;
        lastGestureMs = now;
        Serial.printf("touch tap: x=%d, page=%u\n", startX, page + 1);
      }
    }
  }
}

static void logicTask(void *) {
  uint32_t lastConnectMs = 0, lastPollMs = 0;
  for (;;) {
    uint32_t now = millis();
    if (otaToggleRequested) { otaToggleRequested = false; otaActive ? stopOtaPortal() : startOtaPortal(); }
    if (otaActive) otaServer.handleClient();
    if ((!client || !client->isConnected()) && now - lastConnectMs >= 6000) { lastConnectMs = now; findAndConnectBattery(); }
    if (batteryConnected && now - lastPollMs >= 250) { lastPollMs = now; pollBattery(); }
    vTaskDelay(pdMS_TO_TICKS(20));
  }
}
static void uiTask(void *) {
  for (;;) {
    serviceTouch();
    if (frameDirty) { frameDirty = false; dataDirty = false; drawFullPage(snapshotTelemetry()); }
    else if (dataDirty) { dataDirty = false; drawChangedValues(snapshotTelemetry()); }
    vTaskDelay(pdMS_TO_TICKS(15));
  }
}

void setup() {
  Serial.begin(115200); delay(250); Serial.println("PAC-BATTERY boot; BMS access is read-only");

  // Latch the Waveshare board's battery supply immediately. This is GPIO 7
  // (PWR_Control_PIN) in the vendor PWR_Key example; it only controls this
  // dashboard's own power hold circuit, never the connected battery BMS.
  pinMode(PWR_KEY_INPUT, INPUT);
  pinMode(PWR_CONTROL, OUTPUT);
  digitalWrite(PWR_CONTROL, HIGH);

  telemetryMutex = xSemaphoreCreateMutex(); uiMutex = xSemaphoreCreateMutex();
  pinMode(LCD_BL, OUTPUT); analogWrite(LCD_BL, 128); // 50% brightness to stop overheating!
  SPI.begin(LCD_SCK, -1, LCD_MOSI, LCD_CS); 
  lcd.init(240, 320); 
  lcd.setRotation(0); 
  lcd.setTextWrap(false); 
  Touch_Init();
  NimBLEDevice::init("LFPGo400 Dashboard"); NimBLEDevice::setPower(ESP_PWR_LVL_P9); Serial.println("display and BLE initialized");
  xTaskCreatePinnedToCore(logicTask, "bms_logic", 8192, nullptr, 1, nullptr, 0);
  xTaskCreatePinnedToCore(uiTask, "display_ui", 8192, nullptr, 1, nullptr, 1);
}
void loop() { vTaskDelay(portMAX_DELAY); }
